Columns: title,url,difficulty,topics — topics are semicolon-separated (e.g. Arrays;Hash Table). A header row is optional.

To generate this from your own solved problems: open any leetcode.com page while logged in, open the browser console (F12 → Console), paste the script below, and copy the CSV text it prints.

(async () => {
  const query = `
    query problemsetQuestionList($categorySlug: String, $skip: Int, $limit: Int, $filters: QuestionListFilterInput) {
      problemsetQuestionList: questionList(
        categorySlug: $categorySlug
        limit: $limit
        skip: $skip
        filters: $filters
      ) {
        total: totalNum
        questions: data {
          title
          titleSlug
          difficulty
          status
          topicTags { name }
        }
      }
    }
  `;

  const rows = [];
  const limit = 100;
  let skip = 0;
  let total = Infinity;

  while (skip < total) {
    const res = await fetch("https://leetcode.com/graphql", {
      method: "POST",
      headers: { "content-type": "application/json" },
      credentials: "include",
      body: JSON.stringify({
        query,
        variables: { categorySlug: "algorithms", skip, limit, filters: {} },
      }),
    });
    const json = await res.json();
    if (json.errors) {
      console.error("GraphQL errors:", json.errors);
      return;
    }
    const page = json.data.problemsetQuestionList;
    total = page.total;
    for (const q of page.questions) {
      if (q.status === "ac") rows.push(q);
    }
    skip += limit;
  }

  const csvLines = rows.map(q => {
    const url = `https://leetcode.com/problems/${q.titleSlug}/`;
    const topics = q.topicTags.map(t => t.name).join(";");
    const title = q.title.includes(",") ? `"${q.title}"` : q.title;
    return `${title},${url},${q.difficulty},${topics}`;
  });

  const csv = "title,url,difficulty,topics\n" + csvLines.join("\n");
  console.log(csv);
  try {
    await navigator.clipboard.writeText(csv);
    console.log(`Copied ${rows.length} solved problems to clipboard.`);
  } catch {
    console.log(`${rows.length} rows — clipboard write blocked, copy the logged text above manually.`);
  }
})();
